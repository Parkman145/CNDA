cmake_minimum_required(VERSION 3.31.5)

project(CNDA LANGUAGES CXX)

add_library(CNDA INTERFACE)

target_include_directories(CNDA INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

target_compile_features(CNDA INTERFACE cxx_std_23)

add_library(CNDA::CNDA ALIAS CNDA)

option(BUILD_MAIN "Build main executable" ON)

if (BUILD_MAIN)
    add_executable(cnda_main main.cpp)
    target_link_libraries(cnda_main PRIVATE CNDA)
endif()
